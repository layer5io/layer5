The **Permissions** module in Sistent provides declarative authorization abstractions that integrate directly with `@meshery/schemas/permissions` and host ability evaluators.

<a id="PermissionProvider-Setup">
  <h2>1. Mounting PermissionProvider</h2>
</a>

Mount `PermissionProvider` at the root of your application. Supply the authorization check function (`userHasPermission`) and the user's active session metadata (`userContext`).

<div className="showcase">
  <div className="items" style={{ display: "flex", justifyContent: "center", width: "100%" }}>
    <ThemeWrapper>
      <ProviderStateDemo />
    </ThemeWrapper>
  </div>
  <CodeBlock name="permission-provider-setup" collapsible code={`import { PermissionProvider } from "@sistent/sistent";
import { canKey } from "@/utils/can";

export const App = ({ children, activeUser, currentOrg }) => {
  const userContext = {
    userName: activeUser.firstName + " " + activeUser.lastName,
    orgName: currentOrg?.name,
    roleNames: activeUser.assignedRoles // Assigned roles, not catalog
  };

  return (
    <PermissionProvider userHasPermission={canKey} userContext={userContext}>
      {children}
    </PermissionProvider>
  );
};`} />
</div>

<a id="Native-Button-Gating">
  <h2>2. Native Gating: Button & IconButton</h2>
</a>

Pass `permissionKey` and `permissionAction` directly to `Button` or `IconButton`. When unauthorized, `permissionAction="showShield"` automatically wraps the button in `PermissionShield`, while `permissionAction="hide"` renders `null`.

<div className="showcase">
  <div className="items" style={{ display: "flex", justifyContent: "center", width: "100%" }}>
    <ThemeWrapper>
      <NativeButtonsDemo />
    </ThemeWrapper>
  </div>
  <CodeBlock name="native-button-gating" collapsible code={`import { Button, IconButton } from "@sistent/sistent";
import DeleteIcon from "@mui/icons-material/Delete";
import { Keys } from "@meshery/schemas/permissions";

export const ActionToolbar = () => (
  <div style={{ display: "flex", gap: "12px", alignItems: "center" }}>
    {/* Button with showShield (inline shield on deny) */}
    <Button
      variant="contained"
      color="primary"
      permissionKey={Keys.WorkspaceManagementCreateWorkspace}
      permissionAction="showShield"
    >
      Create Workspace
    </Button>

    {/* IconButton with showShield (badge shield on deny) */}
    <IconButton
      color="error"
      aria-label="Delete Workspace"
      permissionKey={Keys.WorkspaceManagementDeleteWorkspace}
      permissionAction="showShield"
    >
      <DeleteIcon />
    </IconButton>

    {/* Button with hide (renders null on deny) */}
    <Button
      variant="outlined"
      permissionKey={Keys.IdentityAccessManagementEditOrganization}
      permissionAction="hide"
    >
      Org Settings
    </Button>
  </div>
);`} />
</div>

<a id="useAccessibleOrgs">
  <h2>7. Multi-Tenant Hook: useAccessibleOrgs</h2>
</a>

Use `useAccessibleOrgs` to query the user's alternative organizations in parallel and populate an organization switcher on 403 error pages.

<div className="showcase">
  <div className="items" style={{ display: "flex", justifyContent: "center", width: "100%" }}>
    <ThemeWrapper>
      <OrgSwitcherDemo />
    </ThemeWrapper>
  </div>
  <CodeBlock name="use-accessible-orgs-hook" collapsible code={`import { useAccessibleOrgs } from "@sistent/sistent";
import { useGetActiveOrgs, useLazyGetUserKeysQuery } from "@/api/identity";
import { useSelector } from "react-redux";

export const OrgSwitcher = ({ permissionKey }) => {
  const { data: allOrgs, isSuccess: orgsLoaded } = useGetActiveOrgs();
  const currentOrg = useSelector((state) => state.organization.currentOrg);
  const [triggerGetKeys] = useLazyGetUserKeysQuery();

  const { accessibleOrgs, isLoading, hasNoAccessibleOrgs } = useAccessibleOrgs({
    allOrgs: allOrgs ?? [],
    currentOrgId: currentOrg?.id,
    orgsLoaded,
    permissionKey,
    triggerGetKeys
  });

  if (isLoading) return <div>Checking accessible organizations...</div>;
  if (hasNoAccessibleOrgs) return <div>No other organization grants this permission.</div>;

  return (
    <select onChange={(e) => switchOrg(e.target.value)}>
      {accessibleOrgs.map((org) => (
        <option key={org.id} value={org.id}>{org.name}</option>
      ))}
    </select>
  );
};`} />
</div>
