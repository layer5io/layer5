  `// 6. Multi-Tenant Organization Switching with useAccessibleOrgs
import { useAccessibleOrgs } from "@sistent/sistent";
import { useGetActiveOrgs, useLazyGetUserKeysQuery } from "@/api/identity";
import { useSelector } from "react-redux";

export const OrgSwitcher = ({ permissionKey }) => {
  const { data: allOrgs, isSuccess: orgsLoaded } = useGetActiveOrgs();
  const currentOrg = useSelector((state) => state.organization.currentOrg);
  const [triggerGetKeys] = useLazyGetUserKeysQuery();

  const { accessibleOrgs, isLoading, hasNoAccessibleOrgs } = useAccessibleOrgs({
    allOrgs: allOrgs ?? [],
    currentOrgId: currentOrg?.id,
    orgsLoaded,
    permissionKey,
    triggerGetKeys
  });

  if (isLoading) return <div>Checking accessible organizations...</div>;
  if (hasNoAccessibleOrgs) return <div>No other organization grants this permission.</div>;

  return (
    <select onChange={(e) => switchOrg(e.target.value)}>
      {accessibleOrgs.map((org) => (
        <option key={org.id} value={org.id}>{org.name}</option>
      ))}
    </select>
  );
};`

The **Permission System** in Sistent is designed to be completely framework-agnostic while providing deep integration with `@meshery/schemas/permissions` and host ability engines like CASL.

<h3>PermissionShieldProps</h3>

| Prop | Type | Default | Description |
| :--- | :--- | :--- | :--- |
| `permissionKey` | `PermissionKeySpec` | _required_ | Single `Key` or `{ anyOf: Key[] }` / `{ allOf: Key[] }` set. |
| `children` | `React.ReactNode` | _required_ | The affordance to guard. |
| `variant` | `'inline' \| 'badge'` | `'inline'` | Visual placement of the shield trigger icon. |

<h3>PermissionSessionContextProps</h3>

| Prop | Type | Default | Description |
| :--- | :--- | :--- | :--- |
| `variant` | `'tooltip' \| 'card'` | `'tooltip'` | Visual theme and sizing variant. |
| `permissionKey` | `PermissionKeySpec` | `undefined` | Self-resolving key or key set. |
| `displayedKeys` | `Key[]` | `undefined` | Pre-resolved array of keys to render. |
| `subtitle` | `string` | `undefined` | Custom requirement subtitle. |
| `categories` | `string[]` | `undefined` | Category chip labels. |
| `subcategories` | `string[]` | `undefined` | Subcategory chip labels. |

<h3>PermissionProviderProps</h3>

| Prop | Type | Default | Description |
| :--- | :--- | :--- | :--- |
| `userHasPermission` | `(key: Key) => boolean` | _required_ | Authorization evaluator function. |
| `userContext` | `PermissionUserContext` | `undefined` | Session metadata: `{ userName, orgName, roleNames }`. |
| `children` | `React.ReactNode` | _required_ | Root application sub-tree. |
