import { useState } from "react";
import { Box, Button, BottomSheet, Typography } from "@sistent/sistent";

export const BasicBottomSheetCode = () => {
  const [open, setOpen] = useState(false);
  return (
    <>
      <Button variant="contained" onClick={() => setOpen(true)}>Open Basic BottomSheet</Button>
      <BottomSheet open={open} onClose={() => setOpen(false)} title="Basic BottomSheet">
        <Typography>This is a basic bottom sheet with simple content for displaying information.</Typography>
      </BottomSheet>
    </>
  );
};

export const CustomHeightBottomSheetCode = () => {
  const [open, setOpen] = useState(false);
  return (
    <>
      <Button variant="contained" color="secondary" onClick={() => setOpen(true)}>Open Custom Height BottomSheet</Button>
      <BottomSheet open={open} onClose={() => setOpen(false)} title="Tall BottomSheet" maxHeight="95vh">
        <Box sx={{ p: 2 }}>
          <Typography variant="body1" paragraph>
            This bottom sheet is configured with a <code>maxHeight</code> of <code>"95vh"</code>.
          </Typography>
          <Typography variant="body2" color="textSecondary">
            Adjusting the maximum height is useful when you need to display larger amounts of content, complex forms, or scrolling lists.
          </Typography>
        </Box>
      </BottomSheet>
    </>
  );
};

export const CustomHeaderBottomSheetCode = () => {
  const [open, setOpen] = useState(false);
  return (
    <>
      <Button variant="contained" onClick={() => setOpen(true)}>Open Styled Header BottomSheet</Button>
      <BottomSheet
        open={open}
        onClose={() => setOpen(false)}
        title="Custom Header"
        headerBackgroundColor="#1976d2"
        headerTextColor="#ffffff"
      >
        <Box sx={{ p: 2 }}>
          <Typography>
            This bottom sheet uses custom background and text colors for its header.
          </Typography>
        </Box>
      </BottomSheet>
    </>
  );
};

export const NoTitleBottomSheetCode = () => {
  const [open, setOpen] = useState(false);
  return (
    <>
      <Button variant="contained" onClick={() => setOpen(true)}>Open Without Title</Button>
      <BottomSheet open={open} onClose={() => setOpen(false)}>
        <Box sx={{ textAlign: "center", p: 4 }}>
          <Typography variant="h6" gutterBottom>Custom Layout</Typography>
          <Typography paragraph>
            Omitting the title prop removes the default header block entirely, allowing you to build completely custom layouts within the bottom sheet.
          </Typography>
          <Button variant="outlined" onClick={() => setOpen(false)}>Dismiss</Button>
        </Box>
      </BottomSheet>
    </>
  );
};
